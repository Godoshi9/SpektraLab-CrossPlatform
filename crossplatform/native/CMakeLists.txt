cmake_minimum_required(VERSION 3.22)
project(SpektraLabCrossPlatform LANGUAGES C CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  add_compile_options(/utf-8 /bigobj)
  add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
  add_link_options(/MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/utf8.manifest")
endif()
set(SPEKTRALAB_BUILD_NATIVE_RAW ON CACHE BOOL "" FORCE)
set(SPEKTRALAB_BUILD_DESKTOP OFF CACHE BOOL "" FORCE)
set(SPEKTRALAB_BUILD_HOST OFF CACHE BOOL "" FORCE)
set(SPEKTRALAB_LIBRAW_OPENMP OFF CACHE BOOL "" FORCE)
set(SPEKTRALAB_REQUIRE_VULKAN ON CACHE BOOL "" FORCE)
set(SPEKTRALAB_BUILD_TEST_DLL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../engine engine)
add_library(spektralab_mobile SHARED bridge.cpp)
target_include_directories(spektralab_mobile PRIVATE ../../engine/include ../../engine/src)
target_link_libraries(spektralab_mobile PRIVATE spektralab_engine spk_image_io)
if(WIN32)
  # This single-photo host admits a bounded working image and trims idle
  # buffers. Keep an actual 1 GiB safety floor instead of rejecting even a
  # tiny photo whenever Windows has less than the full desktop's 3 GiB free.
  target_compile_definitions(spektralab_engine PRIVATE SPK_WINDOWS_COMMIT_RESERVE_MIB=1024)
  target_compile_definitions(spektralab_mobile PRIVATE SPK_WINDOWS_COMMIT_RESERVE_MIB=1024)
endif()
if(ANDROID)
  target_sources(spektralab_mobile PRIVATE jni.cpp)
  target_link_options(spektralab_mobile PRIVATE -Wl,-z,max-page-size=16384)
endif()
if(BUILD_TESTING)
  add_executable(spk_crossplatform_test ../tests/native_test.cpp)
  target_link_libraries(spk_crossplatform_test PRIVATE spektralab_mobile)
  target_include_directories(spk_crossplatform_test PRIVATE .)
  add_test(NAME spk_crossplatform COMMAND spk_crossplatform_test "${CMAKE_CURRENT_BINARY_DIR}/engine/resources")
  add_executable(spk_import_budget_test ../tests/import_budget_test.cpp)
  target_include_directories(spk_import_budget_test PRIVATE . ../../engine/src)
  target_compile_definitions(spk_import_budget_test PRIVATE SPK_WINDOWS_COMMIT_RESERVE_MIB=1024)
  add_test(NAME spk_import_budget COMMAND spk_import_budget_test)
endif()
